Quick Start

Learn Jaunty basics in 5 minutes.

Time: 5 minutes

This guide covers the essential Jaunty operations: Query, Insert, Update, and Delete.

Step 1: Define Your Entity

public class Product
{
    public int Id { get; set; }
    public string Name { get; set; }
    public decimal Price { get; set; }
    public int CategoryId { get; set; }
}

Step 2: Set Up Connection

using System.Data.SQLite;
using Jaunty;

using var connection = new SQLiteConnection("Data Source=products.db");
connection.Open();

Step 3: Query Data

Basic Query

// Get all products
var products = connection.Query<Product>("SELECT * FROM products");

Query with Parameters

// Named parameters (recommended)
var products = connection.Query<Product>(
    "SELECT * FROM products WHERE category_id = @CategoryId",
    new { CategoryId = 5 });

// Positional parameters
var products = connection.Query<Product>(
    "SELECT * FROM products WHERE category_id = @Id",
    5);

Single Result

// Get first product (throws if none found)
var product = connection.QueryFirst<Product>(
    "SELECT * FROM products WHERE id = @Id",
    new { Id = 1 });

// Get first or null
var product = connection.QueryFirstOrDefault<Product>(
    "SELECT * FROM products WHERE id = @Id",
    new { Id = 1 });

Step 4: Insert Data

var newProduct = new Product
{
    Name = "Widget",
    Price = 19.99m,
    CategoryId = 5
};

// Insert returns the new identity value
var newId = connection.Insert(newProduct);
Console.WriteLine($"Created product with ID: {newId}");

Step 5: Update Data

// Update requires [Key] attribute or Id property
product.Price = 24.99m;
var rowsAffected = connection.Update(product);
Console.WriteLine($"Updated {rowsAffected} rows");

Step 6: Delete Data

// Delete by entity
var rowsAffected = connection.Delete(product);

// Delete by ID
rowsAffected = connection.Delete<Product>(1);

Async Example

// All methods have async variants
var products = await connection.QueryAsync<Product>(
    "SELECT * FROM products",
    cancellationToken: CancellationToken.None);

var newId = await connection.InsertAsync(newProduct);
